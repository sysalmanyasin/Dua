  await page.screenshot({ path: '/tmp/shots/07-formulas.png' });
  // tap a formula cell: bar shows the formula and what it shows

  out.saveAndRedrawMs = Date.now() - t1;
  await page.evaluate(() => { document.querySelector('.ws-grid-scroll').scrollLeft = 96 * 6; }); await page.waitForTimeout(300);

  out.scroll = await page.evaluate(async () => { const s = document.querySelector('.ws-grid-scroll'); const times = []; for (let i = 0; i < 60; i += 1) { const a = performance.now(); s.scrollTop = 28 * (100 + i * 9); await new Promise((r) => requestAnimationFrame(() => requestAnimationFrame(r))); times.push(performance.now() - a); } times.sort((x, y) => x - y); return { median_ms: +times[30].toFixed(1), p95_ms: +times[56].toFixed(1) }; });

  console.log(JSON.stringify(out, null, 1));
